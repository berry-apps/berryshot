# Add BerryShot MCP to Claude Code with one command:
claude mcp add berryshot -- /Applications/BerryShot.app/Contents/Helpers/BerryShotMCP
# Codex config.toml configuration
[mcp_servers.berryshot]
command = "/Applications/BerryShot.app/Contents/Helpers/BerryShotMCP"
startup_timeout_sec = 10
tool_timeout_sec = 60
required = false
// .mcp.json or ~/.claude.json
{
  "mcpServers": {
    "berryshot": {
      "type": "stdio",
      "command": "/Applications/BerryShot.app/Contents/Helpers/BerryShotMCP",
      "args": []
    }
  }
}
# Upload screenshot to your custom HTTP API webhook
curl -X POST "https://your-api.com/v1/screenshots/upload" \
